cmake_minimum_required(VERSION 3.24)
project(Llamatron VERSION 3.0.0 LANGUAGES C CXX RC)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
# The C/C++ runtime linked statically: the portable folder is just the exe.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Release builds keep a PDB (crash dumps, profiles).
if(WIN32 AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  string(APPEND CMAKE_C_FLAGS_RELEASE " -gcodeview -gline-tables-only")
  string(APPEND CMAKE_CXX_FLAGS_RELEASE " -gcodeview -gline-tables-only")
  string(APPEND CMAKE_EXE_LINKER_FLAGS_RELEASE " -Wl,-debug -Wl,-opt:ref -Wl,-opt:icf")
endif()

# SDL3, static.
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/SDL EXCLUDE_FROM_ALL)

# Musashi 68000 core: its sources are copied to the build dir with our m68kconf.h over theirs, and the opcode
# tables are generated by its m68kmake tool.
set(MUSASHI_SRC ${CMAKE_SOURCE_DIR}/third_party/musashi)
set(MUSASHI_GEN ${CMAKE_BINARY_DIR}/musashi)
file(MAKE_DIRECTORY ${MUSASHI_GEN})
foreach(f m68k.h m68kcpu.h m68kcpu.c m68kdasm.c m68kfpu.c m68kmmu.h m68k_in.c m68kmake.c)
  configure_file(${MUSASHI_SRC}/${f} ${MUSASHI_GEN}/${f} COPYONLY)
endforeach()
file(COPY ${MUSASHI_SRC}/softfloat DESTINATION ${MUSASHI_GEN})
configure_file(${CMAKE_SOURCE_DIR}/src/musashi/m68kconf.h ${MUSASHI_GEN}/m68kconf.h COPYONLY)

add_executable(m68kmake ${MUSASHI_GEN}/m68kmake.c)
target_compile_definitions(m68kmake PRIVATE _CRT_SECURE_NO_WARNINGS)
add_custom_command(
  OUTPUT ${MUSASHI_GEN}/m68kops.c ${MUSASHI_GEN}/m68kops.h
  COMMAND m68kmake ${MUSASHI_GEN} ${MUSASHI_GEN}/m68k_in.c
  DEPENDS m68kmake ${MUSASHI_GEN}/m68k_in.c
  WORKING_DIRECTORY ${MUSASHI_GEN})

add_library(musashi STATIC
  ${MUSASHI_GEN}/m68kcpu.c ${MUSASHI_GEN}/m68kdasm.c ${MUSASHI_GEN}/m68kops.c
  ${MUSASHI_GEN}/softfloat/softfloat.c ${CMAKE_SOURCE_DIR}/src/musashi_state.c)
target_include_directories(musashi PUBLIC ${MUSASHI_GEN})
target_compile_definitions(musashi PRIVATE _CRT_SECURE_NO_WARNINGS)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(musashi PRIVATE -w)
endif()

configure_file(res/llamatron.rc.in ${CMAKE_BINARY_DIR}/llamatron.rc @ONLY)
add_executable(llamatron WIN32
  ${CMAKE_BINARY_DIR}/llamatron.rc
  src/main.cpp
  src/machine.cpp
  src/cpu_musashi.cpp
  src/ym2149.cpp
  src/disk.cpp
  src/input.cpp
  src/game.cpp
  src/gridrunner.cpp
  src/tripatron.cpp
  src/music.cpp
  src/miniaudio_impl.cpp
  src/launcher.cpp
  src/games.cpp
  src/pad.cpp
  src/llamazap.cpp
  src/remap.cpp
  src/stgames.cpp
  src/crt.cpp
  src/remaster.cpp
  src/remaster_andes.cpp
  src/remaster_llama.cpp
  src/remaster_grid.cpp
  src/remaster_rmc.cpp
  src/remaster_d2.cpp
  src/remaster_falcon.cpp
  src/savestate.cpp
  src/settings.cpp
  src/settings_menu.cpp
  src/steam.cpp
  src/external.cpp
  src/preview.cpp
  src/rumble.cpp
  src/glyphs.cpp
  src/log.cpp
  src/crash.cpp)
target_link_libraries(llamatron PRIVATE SDL3::SDL3-static musashi)
# miniaudio 0.11.21 (public domain / MIT-0), as in our Neon port: loopback capture of the PC's sound.
target_include_directories(llamatron PRIVATE ${CMAKE_SOURCE_DIR}/third_party/miniaudio)
set_source_files_properties(src/miniaudio_impl.cpp PROPERTIES COMPILE_OPTIONS "-w")
target_compile_definitions(llamatron PRIVATE _CRT_SECURE_NO_WARNINGS)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(llamatron PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

# Defender 2000 (Atari Jaguar; moved here from the Tempest collection 2026-11-22): d2k.exe, its own program on the
# Virtual Jaguar core (GPL-3, vendored in third_party/virtualjaguar with our patches - so d2k.exe is GPL-3; the ST
# launcher starts it as a separate process). Notes: D2K.md.
set(VJ_LTO_ENABLED OFF)
function(vj_optimise target)
endfunction()
list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/third_party/virtualjaguar/app/cmake")
include(VJCore)
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
  target_compile_options(vj_core PRIVATE -w)
  target_compile_options(vj_libchdr PRIVATE -w)
endif()
configure_file(res/d2k.rc.in ${CMAKE_BINARY_DIR}/d2k.rc @ONLY)
add_executable(d2k WIN32
  ${CMAKE_BINARY_DIR}/d2k.rc
  src/d2k/main.cpp src/d2k/host.cpp src/d2k/settings.cpp src/d2k/volume.cpp src/d2k/runlog.cpp src/d2k/menu.cpp
  src/d2k/pixfont.cpp src/d2k/d2k_menu.cpp src/d2k/controls.cpp src/d2k/traps.cpp src/d2k/gpulog.cpp
  src/d2k/jagsurvey.cpp src/d2k/d2k.cpp src/d2k/d2k_bonus.cpp src/d2k/d2k_stars.cpp src/d2k/perf.cpp src/d2k/xbr.cpp
  src/d2k/crt.cpp src/d2k/crt_menu.cpp)
target_link_libraries(d2k PRIVATE vj_core SDL3::SDL3-static)
target_compile_definitions(d2k PRIVATE _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
if(WIN32)
  target_link_libraries(d2k PRIVATE ws2_32 iphlpapi winmm)
endif()
# The high-res art pack (Tools/d2k/d2k_art_pack.py; made from the cartridge's pictures, so kept in Private\, not in
# git) next to the exe, when it's there.
if(EXISTS "${CMAKE_SOURCE_DIR}/Private/d2k/d2k_art/pack.txt")
  add_custom_command(TARGET d2k POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/Private/d2k/d2k_art" $<TARGET_FILE_DIR:d2k>/d2k_art)
endif()
